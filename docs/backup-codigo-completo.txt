# Guia para agentes neste repositório

Este repositório roda numa plataforma híbrida: **Skip** (no-code/AI builder) +
**git** (GitHub) + **PocketBase** (backend self-hosted pelo Skip).

## Stack
- **Front:** React + Vite + TypeScript + shadcn/ui + Tailwind. Entrada em `src/main.tsx`, rotas em `src/App.tsx`.
- **Backend:** PocketBase JSVM (Goja). Hooks em `pocketbase/hooks/*.js`, migrations em `pocketbase/migrations/*.js`.
- **Integração externa:** Evolution API (WhatsApp Baileys) chamada via `$http.send` dos hooks.
- **Plataforma:** o Skip Cloud roda o PocketBase, gerencia sync com GitHub e faz publish/build do front.

import { ClientResponseError } from 'pocketbase'

export type FieldErrors = Record<string, string>

export function extractFieldErrors(err: unknown): FieldErrors {
  if (err instanceof ClientResponseError && err.data?.data) {
    const fieldErrors: FieldErrors = {}
    for (const [key, value] of Object.entries(err.data.data)) {
      if (typeof value === 'object' && value !== null && 'message' in value) {
        fieldErrors[key] = String((value as { message: unknown }).message)
      } else if (typeof value === 'string') {
        fieldErrors[key] = value
      }
    }
    return fieldErrors
  }
  return {}
}

export function getErrorMessage(err: unknown, fallback = 'Ocorreu um erro.'): string {
  if (err instanceof ClientResponseError) {
    if (err.message && !err.message.includes('Something went wrong')) {
      return err.message
    }
  }
  if (err instanceof Error) return err.message
  return fallback
}

import React, { createContext, useContext, useEffect, useState } from 'react'
import { pb } from '@/lib/pocketbase/client'
import { User } from '@/types/models'

interface AuthContextType {
  user: User | null
  token: string | null
  isLoading: boolean
  logout: () => void
}

const AuthContext = createContext<AuthContextType>({
  user: null,
  token: null,
  isLoading: true,
  logout: () => {},
})

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<User | null>(null)
  const [token, setToken] = useState<string | null>(null)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    const updateUser = () => {
      if (pb.authStore.isValid && pb.authStore.record) {
        setUser({
          id: pb.authStore.record.id,
          email: pb.authStore.record.email || '',
          name: pb.authStore.record.name || '',
          avatar: pb.authStore.record.avatar || '',
        })
        setToken(pb.authStore.token)
      } else {
        setUser(null)
        setToken(null)
      }
      setIsLoading(false)
    }

    updateUser()

    const unsubscribe = pb.authStore.onChange(() => {
      updateUser()
    })

    return () => {
      unsubscribe()
    }
  }, [])

  const logout = () => {
    pb.authStore.clear()
    setUser(null)
    setToken(null)
  }

  return (
    <AuthContext.Provider value={{ user, token, isLoading, logout }}>
      {children}
    </AuthContext.Provider>
  )
}

export const useAuth = () => useContext(AuthContext)

import { useEffect, useRef } from 'react'
import { pb } from '@/lib/pocketbase/client'
import { RecordSubscription } from 'pocketbase'

export function useRealtime<T = any>(
  collection: string,
  callback: (data: RecordSubscription<T>) => void,
  filter?: string,
) {
  const callbackRef = useRef(callback)
  callbackRef.current = callback

  useEffect(() => {
    let isSubscribed = true

    const subscribe = async () => {
      try {
        await pb.collection(collection).subscribe<T>(
          filter || '*',
          (e) => {
            if (isSubscribed) {
              callbackRef.current(e)
            }
          },
        )
      } catch (err) {
        console.warn(`Erro ao assinar realtime para ${collection}:`, err)
      }
    }

    subscribe()

    return () => {
      isSubscribed = false
      pb.collection(collection).unsubscribe(filter || '*').catch(() => {})
    }
  }, [collection, filter])
}

import { useState, useEffect } from 'react'
import { pb } from '@/lib/pocketbase/client'
import { useAuth } from './use-auth'
import { Account, AccountMember } from '@/types/models'

export function useCurrentAccount() {
  const { user } = useAuth()
  const [account, setAccount] = useState<Account | null>(null)
  const [membership, setMembership] = useState<AccountMember | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (!user) {
      setAccount(null)
      setMembership(null)
      setLoading(false)
      return
    }

    let isMounted = true

    async function loadAccount() {
      try {
        const member = await pb
          .collection('account_members')
          .getFirstListItem<AccountMember>(`user_id = "${user?.id}"`, {
            expand: 'account_id',
          })

        if (!isMounted) return

        setMembership(member)
        if (member && (member as any).expand?.account_id) {
          setAccount((member as any).expand.account_id)
        } else if (member?.account_id) {
          const acc = await pb.collection('accounts').getOne<Account>(member.account_id)
          if (isMounted) setAccount(acc)
        }
      } catch (err) {
        console.warn('Erro ao carregar conta atual:', err)
      } finally {
        if (isMounted) setLoading(false)
      }
    }

    loadAccount()

    return () => {
      isMounted = false
    }
  }, [user])

  return { account, membership, loading, isOwner: membership?.role === 'owner' }
}
